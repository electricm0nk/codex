//! E8.MC probe (not committed as a test; kept as artifacts/epic_8/E8.MC/probe.rs.txt).
use crate::character_hub::{load_saved_character_at_root, seed_default_characters_at};
use crate::print::{render_sheet_pdf_at_root, sheet_document_at_root, Paper, PrintRequest};
use super::skills::skill_name;
use std::fmt::Write;

fn text(bytes: &[u8]) -> String {
    pdf_extract::extract_text_from_mem(bytes).unwrap().split_whitespace().collect::<Vec<_>>().join(" ")
}
fn signed(v: i16) -> String { if v >= 0 { format!("+{v}") } else { v.to_string() } }

#[test]
fn mc_probe_pf_screen_values_in_basic_pdf() {
    let out_dir = std::env::var("MC_PROBE_OUT").expect("MC_PROBE_OUT");
    let dir = std::env::temp_dir().join(format!("codex-mc-probe-{}", std::process::id()));
    let _ = std::fs::remove_dir_all(&dir);
    std::fs::create_dir_all(&dir).unwrap();
    seed_default_characters_at(&dir, "0.0.0-test").unwrap();
    let mut report = String::new();
    for n in 1..=3 {
        let id = format!("00000000-0000-0000-0000-00000000000{n}");
        let root = dir.join("characters").join(&id);
        let loaded = load_saved_character_at_root(&root).unwrap();
        let request = PrintRequest { paper: Paper::A4, spellbook: false, math: false, bio: false, campaigns: Vec::new() };
        let pdf = render_sheet_pdf_at_root(&root, &request).unwrap();
        let t = text(&pdf.pdf);
        std::fs::write(format!("{out_dir}/pf-seed-{n}-basic.txt"), &t).unwrap();
        writeln!(report, "== seed {n} {} warnings={}", loaded.summary.display_label, pdf.warnings.len()).unwrap();
        let mut total = 0; let mut missing = Vec::new();
        for row in loaded.explanations.iter().filter(|r| r.id.starts_with("skill.total.")) {
            total += 1;
            let slug = row.id.trim_start_matches("skill.total.");
            let name = skill_name(slug);
            let needle = format!("{name} ");
            let printed = t.find(&needle).map(|at| t[at + needle.len()..].split(' ').take(2).collect::<Vec<_>>().join(" "));
            let ok = printed.as_deref().is_some_and(|p| p.split(' ').any(|w| w == signed(row.value)));
            if !ok { missing.push(format!("{name}={}", signed(row.value))); }
        }
        writeln!(report, "skill.total rows: {total}; printed with their total: {}; missing: {}", total - missing.len(), missing.len()).unwrap();
        writeln!(report, "  missing: {}", missing.join(", ")).unwrap();
        let mut not_found = Vec::new();
        for row in &loaded.explanations {
            if row.id.starts_with("skill.total.") { continue; }
            let v = row.value.to_string();
            let s = signed(row.value);
            if !t.split(' ').any(|w| w == v || w == s || w.trim_end_matches(['.', ',']) == v) { not_found.push(format!("{}={}", row.id, row.value)); }
        }
        writeln!(report, "other explanation rows: {}; value token absent from basic PDF: {}", loaded.explanations.len() - total, not_found.len()).unwrap();
        for nf in not_found { writeln!(report, "  {nf}").unwrap(); }
        let doc = sheet_document_at_root(&root, &request).unwrap();
        let cited: std::collections::BTreeSet<String> = doc.blocks.iter().flat_map(|b| b.rows.iter()).flat_map(|r| r.derivation.iter().flatten()).map(|d| d.source.split(" = ").next().unwrap_or("").to_owned()).collect();
        let uncited: Vec<String> = loaded.explanations.iter().filter(|r| !cited.contains(&r.id)).map(|r| format!("{}={}", r.id, r.value)).collect();
        writeln!(report, "explanation rows cited by no basic-sheet row: {} of {}", uncited.len(), loaded.explanations.len()).unwrap();
        for u in uncited { writeln!(report, "  uncited {u}").unwrap(); }
        let dr = loaded.snapshot.as_ref().and_then(|s| s.damage_reduction);
        writeln!(report, "damage_reduction on screen: {dr:?}; 'Damage Reduction' printed: {}", t.contains("Damage Reduction")).unwrap();
        writeln!(report, "money printed (PP/GP/Platinum/Gold): {}", ["PP", "GP", "Platinum", "Gold pieces", "Money"].iter().any(|w| t.split(' ').any(|x| x == *w))).unwrap();
        writeln!(report, "size/vision printed: Size={} Vision/Darkvision/Low-Light={}", t.contains("Size"), t.contains("ision")).unwrap();
    }
    std::fs::write(format!("{out_dir}/pf-probe-report.txt"), &report).unwrap();
    println!("{report}");
}
