import random
M=(1<<64)-1
def next_(st):
    st=(st+0x9E3779B97F4A7C15)&M; z=st
    z=((z^(z>>30))*0xBF58476D1CE4E5B9)&M
    z=((z^(z>>27))*0x94D049BB133111EB)&M
    return st,z^(z>>31)
def rl32(v,n): return ((v<<n)|(v>>(32-n)))&0xffffffff
def roll4(seed,place=1):
    st=seed^rl32((place*0x85EBCA6B)&0xffffffff,13)
    t=0
    for _ in range(4):
        zone=M-(M%6)
        while True:
            st,d=next_(st)
            if d<zone: t+=d%6+1;break
    return t
for name,gen in [('sequential',range(0,200000)),('random u32',(random.getrandbits(32) for _ in range(200000)))]:
    n=h=0
    for s in gen:
        n+=1; h+=roll4(s)>=21
    print(name,n,h,round(h/n,4))
print('roll seed38',roll4(38),'seed0',roll4(0))
