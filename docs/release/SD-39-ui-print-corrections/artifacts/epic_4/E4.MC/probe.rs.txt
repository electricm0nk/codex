    fn probe_flat(v: &serde_json::Value, path: String, out: &mut std::collections::BTreeMap<String, String>) {
        match v {
            serde_json::Value::Object(m) => { for (k, x) in m { probe_flat(x, format!("{path}/{k}"), out); } }
            serde_json::Value::Array(a) => {
                for (i, x) in a.iter().enumerate() {
                    let key = x.get("id").or_else(|| x.get("itemId")).and_then(|k| k.as_str()).map(str::to_owned).unwrap_or_else(|| i.to_string());
                    probe_flat(x, format!("{path}/{key}"), out);
                }
            }
            other => { out.insert(path, other.to_string()); }
        }
    }
    fn probe_read(root: &Path) -> std::collections::BTreeMap<String, String> {
        let mut out = Default::default();
        probe_flat(&serde_json::to_value(load_saved_character_at_root(root).unwrap()).unwrap(), String::new(), &mut out);
        out
    }
    fn probe_diff(label: &str, a: &std::collections::BTreeMap<String, String>, b: &std::collections::BTreeMap<String, String>) {
        eprintln!("=== {label}");
        for (k, v) in b {
            if k.contains("/detail") || k.contains("/summary/") { continue; }
            if a.get(k) != Some(v) { eprintln!("  {k}: {:?} -> {v}", a.get(k)); }
        }
        for k in a.keys() { if !b.contains_key(k) && !k.contains("/detail") { eprintln!("  {k}: gone"); } }
    }

        ] {
            for item in items {
                let dir = tempdir("e4mc");
                seed_default_characters_at(&dir, "0.0.0-test").unwrap();
                let root = characters_root_from_app_data_dir(&dir).join(seed);
                let a = probe_read(&root);
                worn_vs_carried_hold(&root, item, ActiveState::SelectedInactive);
                let carried = probe_read(&root);
                probe_diff(&format!("{seed} carried {item}"), &a, &carried);

                probe_diff(&format!("{seed} worn {item}"), &carried, &b);
                std::fs::remove_dir_all(&dir).ok();
            }
        }
