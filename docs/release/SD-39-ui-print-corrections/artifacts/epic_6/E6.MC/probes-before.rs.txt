    fn mcp_show(label: &str, r: Result<CreateCharacterResponse, String>) {
        match r {
            Ok(CreateCharacterResponse::Saved { .. }) => println!("MCP {label}: Saved"),
            Ok(CreateCharacterResponse::Blocked { diagnostics }) => println!("MCP {label}: BLOCKED {:?}", diagnostics.iter().map(|d| (d.id.clone(), d.message.chars().take(90).collect::<String>())).collect::<Vec<_>>()),
            Ok(other) => println!("MCP {label}: other {:?}", std::mem::discriminant(&other)),
            Err(e) => println!("MCP {label}: ERR {}", e.chars().take(160).collect::<String>()),
        }
    }
    fn mcp_sheet(label: &str, root: &Path) {
        let loaded = load_saved_character_at_root(root).expect("loads");
        let blocking: Vec<_> = loaded.diagnostics.iter().filter(|d| d.claim_blocking).map(|d| (d.id.clone(), d.message.chars().take(100).collect::<String>())).collect();
        println!("MCP {label}: snapshot={} blocking={blocking:?} admitted={:?}", loaded.snapshot.is_some(), loaded.admitted_under_override.iter().map(|r| (r.kind, r.id.clone(), r.override_flag)).collect::<Vec<_>>());
    }
    fn mcp_class_char(label: &str, race: &str, class: &str, level: u8) -> PathBuf {
        let root = tempdir(label);
        let mut envelope = level_up_test_envelope(race, level);
        envelope.character_input = compose_character_input(&request_for_class(race, class, level));
        let (choices, spells) = codex::rules_core::class_seeds::canonical_seeds_for(class.strip_prefix("class:").unwrap(), level);
        envelope.character_input.chosen.selected_choices.extend(choices);
        envelope.character_input.chosen.spells_selected.extend(spells);
        SavedCharacterStore::save(&envelope, &root).expect("save");
        root
    }
