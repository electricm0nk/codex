## Commands (second implementation)

`members.py` (run with `python3 -I members.py <tree>`):

```python
import json, glob, sys
root = sys.argv[1]
rules = []
for f in glob.glob(root + '/data/sheet_rules/*/*/*.json'):
    rules.extend(json.load(open(f)))
NOT = {"race", "race_trait", "template", "monster", "monster_ability", "trait", "equipment"}
def feat_pool(r):
    p = r['target']['Pool']
    o = r.get('offers'); fr = o and o['from']
    offers_feats = isinstance(fr, dict) and 'Rules' in fr and fr['Rules']['pool'] == 'feat'
    words = p.split('_')
    named = any(w in ('feat', 'feats') for w in words) and 'companion' not in words
    return p == 'feat' or offers_feats or named
offers = []
for r in rules:
    t = r.get('target')
    if not (isinstance(t, dict) and 'Pool' in t): continue
    if feat_pool(r) or r.get('subject') != 'Character' or r['id'].split(':')[1] in NOT: continue
    o = r.get('offers')
    if o and o['from'] not in offers: offers.append(o['from'])
def sel(fr, r):
    if '#' in r['id']: return False
    if isinstance(fr, dict) and 'Rules' in fr:
        tags = [x.lower() for x in r.get('tags', [])]
        return r['pool'] == fr['Rules']['pool'] and all(t.lower() in tags for t in fr['Rules']['tags'])
    if fr == 'Domains': return r['id'].split(':')[1] == 'domain'
    return False
members = [r for r in rules if r.get('label') and any(sel(fr, r) for fr in offers)]
print('offers', len(offers), 'members', len(members), 'distinct ids', len({r['id'] for r in members}))
def terms(a):
    return a['All'] if isinstance(a, dict) and 'All' in a else [a]
print('no prerequisite', sum(1 for r in members if all(t == 'Always' for t in terms(r['applies']))))
print('unconverted rule syntax', sum(1 for r in members if 'only in rule syntax' in json.dumps(r['applies'])))
print('MissingRule members', sum(1 for r in members if 'MissingRule' in json.dumps(r['applies'])))
```
