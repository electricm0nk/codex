#[path = "common/mod.rs"]
mod common;
use codex::rules_core::pilot_compute::build_pilot_headless_receipt;
fn ids(t: &str) -> String {
    let r = build_pilot_headless_receipt(&common::load(t));
    let mut v: Vec<_> = r.computation.diagnostics.iter().filter(|d| d.claim_blocking).map(|d| d.id.clone()).collect();
    v.sort(); v.dedup();
    format!("{:?}|{}", r.status, v.join(","))
}
